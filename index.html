function openSong(type) { const song = getSong(); if (!song) return; const url = type === "karaoke" ? song.karaokeUrl : song.originalUrl; const area = document.getElementById("videoLinkArea"); area.classList.remove("hidden"); area.innerHTML = ""; // Tenta extrair o ID do vídeo do YouTube let videoId = null; const watchMatch = url.match(/[?&]v=([^&]+)/); if (watchMatch) { videoId = watchMatch[1]; } if (videoId) { // Tem link direto → coloca o player dentro do site const iframe = document.createElement("iframe"); iframe.width = "100%"; iframe.height = "315"; iframe.src = `https://www.youtube.com/embed/${videoId}?autoplay=1`; iframe.title = "Player de música"; iframe.frameBorder = "0"; iframe.allow = "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share"; iframe.allowFullscreen = true; iframe.style.borderRadius = "12px"; iframe.style.maxWidth = "100%"; iframe.style.marginTop = "10px"; area.appendChild(iframe); } else { // É busca do YouTube → mostra o link, mas NÃO abre sozinho const label = document.createElement("p"); label.style.marginBottom = "8px"; label.textContent = type === "karaoke" ? "🎤 Busca de karaokê (clique para escolher o vídeo):" : "🎶 Busca da versão original (clique para escolher o vídeo):"; const link = document.createElement("a"); link.href = url; link.target = "_blank"; link.rel = "noopener noreferrer"; link.style.color = "#FFD166"; link.style.fontWeight = "bold"; link.style.display = "inline-block"; link.textContent = "Abrir busca no YouTube"; area.append(label, link); } }
