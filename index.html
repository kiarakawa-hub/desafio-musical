function openSong(type) { const song = getSong(); if (!song) return; const url = type === "karaoke" ? song.karaokeUrl : song.originalUrl; const area = document.getElementById("videoLinkArea"); area.classList.remove("hidden"); area.innerHTML = ""; // Verifica se e um link direto do YouTube let videoId = null; if (url.includes("watch?v=")) { videoId = url.split("watch?v=")[1].split("&")[0]; } if (videoId) { // Coloca o player dentro do site const iframe = document.createElement("iframe"); iframe.width = "100%"; iframe.height = "315"; iframe.src = "https://www.youtube.com/embed/" + videoId + "?autoplay=1"; iframe.title = "Player de musica"; iframe.frameBorder = "0"; iframe.allow = "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"; iframe.allowFullscreen = true; iframe.style.borderRadius = "12px"; iframe.style.maxWidth = "100%"; iframe.style.marginTop = "10px"; area.appendChild(iframe); } else { // Mostra o link sem abrir sozinho const label = document.createElement("p"); label.style.marginBottom = "8px"; if (type === "karaoke") { label.textContent = "Busca de karaoke (clique para escolher o video):"; } else { label.textContent = "Busca da versao original (clique para escolher o video):"; } const link = document.createElement("a"); link.href = url; link.target = "_blank"; link.rel = "noopener noreferrer"; link.style.color = "#FFD166"; link.style.fontWeight = "bold"; link.style.display = "inline-block"; link.textContent = "Abrir busca no YouTube"; area.appendChild(label); area.appendChild(link); } }
